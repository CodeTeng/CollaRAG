# 意图感知的多智能体检索增强系统架构设计

## 🎯 设计目标与动机

### 核心问题

传统的单一Agent RAG系统存在以下局限：

1. **职责过载**：单个Agent需要处理意图识别、检索、推理、生成等所有任务
2. **扩展性差**：新增功能需要修改核心Agent逻辑
3. **并行能力弱**：多个子任务无法真正并行处理
4. **容错能力低**：一个模块失败会影响整个流程

### 多Agent架构的优势

✅ **职责分离**：每个Agent专注于特定任务（意图识别、检索、推理等）  
✅ **并行处理**：多个Agent可以同时工作，提升效率  
✅ **动态协调**：通过消息总线实现Agent间的灵活通信  
✅ **易于扩展**：新增Agent不影响现有系统  
✅ **容错鲁棒**：单个Agent失败不会导致系统崩溃

## 🤖 核心Agent详细设计

### 1. Orchestrator Agent（协调者）

**角色定位**：系统总控，负责任务分发、Agent协调、结果汇总

#### 核心职责

- ✅ 接收用户查询，初始化查询会话
- ✅ 根据查询复杂度动态调度Agent执行顺序
- ✅ 监控所有Agent的执行状态
- ✅ 汇总各Agent的结果，生成最终答案
- ✅ 处理异常和超时情况

#### 关键方法

```python
class OrchestratorAgent:
    """协调者Agent - 系统核心控制器"""

    def __init__(self, message_bus, config):
        self.message_bus = message_bus
        self.config = config
        self.agents = {}  # 注册的所有Agent
        self.session_manager = SessionManager()

    def register_agent(self, agent_name: str, agent_instance):
        """注册Agent到协调者"""
        self.agents[agent_name] = agent_instance
        logger.info(f"Registered agent: {agent_name}")

    async def process_query(self, query: str, user_id: str = None) -> Dict:
        """处理用户查询的主流程"""
        # 创建会话
        session_id = self.session_manager.create_session(user_id, query)

        try:
            # 阶段1: 意图识别
            intent_result = await self.call_agent(
                "intent_agent",
                action="analyze_intent",
                payload={"query": query, "session_id": session_id}
            )

            # 阶段2: 记忆检索（并行）
            memory_result = await self.call_agent(
                "memory_agent",
                action="get_context",
                payload={"session_id": session_id}
            )

            # 阶段3: 策略决策
            strategy_result = await self.call_agent(
                "strategy_agent",
                action="decide_strategy",
                payload={
                    "intent": intent_result,
                    "context": memory_result,
                    "query": query
                }
            )

            # 阶段4: 执行检索（可能触发多个Retrieval Agent）
            retrieval_result = await self.call_agent(
                "retrieval_coordinator",
                action="coordinate_retrieval",
                payload={
                    "query": query,
                    "strategy": strategy_result,
                    "intent": intent_result
                }
            )

            # 阶段5: 推理规划
            reasoning_result = await self.call_agent(
                "reasoning_agent",
                action="plan_reasoning",
                payload={
                    "query": query,
                    "retrieval_result": retrieval_result,
                    "intent": intent_result
                }
            )

            # 阶段6: 生成答案
            response_result = await self.call_agent(
                "response_generator",
                action="generate_answer",
                payload={
                    "query": query,
                    "reasoning": reasoning_result,
                    "retrieval": retrieval_result
                }
            )

            # 阶段7: 反思与评估
            reflection_result = await self.call_agent(
                "reflection_agent",
                action="evaluate_quality",
                payload={
                    "query": query,
                    "answer": response_result,
                    "retrieval": retrieval_result
                }
            )

            # 如果质量不达标，触发重试
            if reflection_result["quality_score"] < self.config.quality_threshold:
                logger.warning(f"Quality score {reflection_result['quality_score']} "
                             f"below threshold, retrying...")
                # 递归调用（最多3次）
                if reflection_result.get("retry_count", 0) < 3:
                    return await self.process_query_with_feedback(
                        query, session_id, reflection_result
                    )

            # 保存到记忆
            await self.call_agent(
                "memory_agent",
                action="save_interaction",
                payload={
                    "session_id": session_id,
                    "query": query,
                    "answer": response_result,
                    "intent": intent_result
                }
            )

            # 返回最终结果
            return {
                "answer": response_result["answer"],
                "intent": intent_result["intent_type"],
                "strategy": strategy_result["strategy_name"],
                "quality_score": reflection_result["quality_score"],
                "retrieval_stats": retrieval_result["stats"],
                "session_id": session_id
            }

        except Exception as e:
            logger.error(f"Error in orchestrator: {e}")
            return {
                "answer": "系统遇到错误，请稍后重试。",
                "error": str(e),
                "session_id": session_id
            }

    async def call_agent(self, agent_name: str, action: str,
                        payload: Dict, timeout: float = 30.0) -> Dict:
        """调用指定Agent执行任务"""
        if agent_name not in self.agents:
            raise ValueError(f"Agent {agent_name} not registered")

        # 通过消息总线发送请求
        message = Message(
            sender="orchestrator",
            receiver=agent_name,
            action=action,
            payload=payload,
            timestamp=time.time()
        )

        # 异步等待响应（带超时）
        response = await self.message_bus.send_and_wait(message, timeout)
        return response.payload

    async def process_query_with_feedback(self, query: str, session_id: str,
                                         feedback: Dict) -> Dict:
        """根据反思反馈重新处理查询"""
        # 提取改进建议
        suggestions = feedback.get("suggestions", [])

        # 调整策略
        adjusted_strategy = await self.call_agent(
            "strategy_agent",
            action="adjust_strategy",
            payload={"feedback": feedback, "suggestions": suggestions}
        )

        # 重新检索
        retrieval_result = await self.call_agent(
            "retrieval_coordinator",
            action="coordinate_retrieval",
            payload={
                "query": query,
                "strategy": adjusted_strategy,
                "retry": True
            }
        )

        # 重新生成答案
        response_result = await self.call_agent(
            "response_generator",
            action="generate_answer",
            payload={
                "query": query,
                "retrieval": retrieval_result,
                "feedback": feedback
            }
        )

        return {
            "answer": response_result["answer"],
            "retry_count": feedback.get("retry_count", 0) + 1,
            "improved": True
        }
```

#### 输入/输出接口

| 方法             | 输入                              | 输出                                                    |
| ---------------- | --------------------------------- | ------------------------------------------------------- |
| `process_query`  | `query: str`, `user_id: str`      | `{answer, intent, strategy, quality_score, session_id}` |
| `call_agent`     | `agent_name`, `action`, `payload` | Agent返回结果                                           |
| `register_agent` | `agent_name`, `agent_instance`    | None                                                    |

### 2. Intent Agent（意图识别Agent）

**角色定位**：专注于查询意图的识别和分类

#### 核心职责

- ✅ 分析用户查询，识别6类意图（factual, analytical, comparative, multi_hop, exploratory, follow_up）
- ✅ 提取查询特征（实体、疑问词、复杂度等）
- ✅ 判断查询是否依赖上下文
- ✅ 计算意图识别的置信度

#### 支持的意图分类

| 意图类型        | 描述     | 示例                                                 | 复杂度     |
| --------------- | -------- | ---------------------------------------------------- | ---------- |
| **factual**     | 事实查询 | "Who is Elon Musk?"                                  | 低         |
| **analytical**  | 分析查询 | "Why did the stock market crash?"                    | 中         |
| **comparative** | 比较查询 | "Compare Python vs Java"                             | 中         |
| **multi_hop**   | 多跳推理 | "Who directed the movie starring Tom Hanks in 1994?" | 高         |
| **exploratory** | 探索查询 | "Tell me about quantum computing"                    | 中         |
| **follow_up**   | 追问     | "What about the second one?"                         | 上下文依赖 |

#### 关键方法

```python
class IntentAgent:
    """意图识别Agent"""

    def __init__(self, message_bus, llm_client):
        self.message_bus = message_bus
        self.llm_client = llm_client
        self.intent_classifier = IntentClassifier(llm_client)

        # 注册消息处理器
        self.message_bus.register_handler("intent_agent", self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息总线的消息"""
        action = message.action
        payload = message.payload

        if action == "analyze_intent":
            result = await self.analyze_intent(
                payload["query"],
                payload.get("session_id")
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="intent_agent",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def analyze_intent(self, query: str, session_id: str = None) -> Dict:
        """分析查询意图"""
        # 1. 规则基础分类（快速初筛）
        rule_based_intent = self._rule_based_classification(query)

        # 2. LLM精确分类（深度理解）
        llm_intent = await self._llm_based_classification(query)

        # 3. 综合判断
        final_intent = self._merge_results(rule_based_intent, llm_intent)

        # 4. 提取查询特征
        features = self._extract_query_features(query)

        # 5. 上下文依赖检测
        requires_context = self._check_context_dependency(query, session_id)

        return {
            "intent_type": final_intent,
            "confidence": features["confidence"],
            "complexity": self._get_complexity(final_intent),
            "features": features,
            "requires_context": requires_context,
            "rule_based_intent": rule_based_intent,
            "llm_intent": llm_intent
        }

    def _rule_based_classification(self, query: str) -> str:
        """基于规则的快速分类"""
        query_lower = query.lower()

        # 疑问词映射
        if any(word in query_lower for word in ["who is", "what is", "when", "where"]):
            return "factual"

        if any(word in query_lower for word in ["why", "how", "explain"]):
            return "analytical"

        if any(word in query_lower for word in ["compare", "difference", "versus", "vs"]):
            return "comparative"

        if any(word in query_lower for word in ["tell me more", "what about"]):
            return "follow_up"

        # 多跳检测（启发式）
        if len(query.split()) > 15 and ("of the" in query_lower or "who" in query_lower):
            return "multi_hop"

        return "exploratory"

    async def _llm_based_classification(self, query: str) -> str:
        """基于LLM的精确分类"""
        prompt = f"""Classify the user's query intent into one of these categories:
- factual: Simple fact queries (Who/What/When/Where)
- analytical: Why/How questions requiring explanation
- comparative: Comparing two or more entities
- multi_hop: Questions requiring multiple reasoning steps
- exploratory: Open-ended exploration
- follow_up: Follow-up or clarification questions

Query: "{query}"

Output only the category name (e.g., "factual").
"""

        response = await self.llm_client.call_async(prompt, max_tokens=20, temperature=0)
        intent = response.strip().lower()

        # 验证输出
        valid_intents = ["factual", "analytical", "comparative", "multi_hop",
                        "exploratory", "follow_up"]
        return intent if intent in valid_intents else "exploratory"

    def _merge_results(self, rule_based: str, llm_based: str) -> str:
        """合并规则和LLM结果"""
        # 如果一致，直接返回
        if rule_based == llm_based:
            return rule_based

        # 优先信任LLM（对于复杂意图）
        complex_intents = ["multi_hop", "analytical", "follow_up"]
        if llm_based in complex_intents:
            return llm_based

        return rule_based

    def _extract_query_features(self, query: str) -> Dict:
        """提取查询特征"""
        import spacy
        nlp = spacy.load("en_core_web_sm")
        doc = nlp(query)

        features = {
            "entity_count": len(doc.ents),
            "entities": [ent.text for ent in doc.ents],
            "entity_types": [ent.label_ for ent in doc.ents],
            "token_count": len([t for t in doc if not t.is_stop]),
            "has_negation": any(t.dep_ == "neg" for t in doc),
            "question_word": self._get_question_word(query),
            "confidence": 0.85  # 可基于规则或模型计算
        }

        return features

    def _get_question_word(self, query: str) -> str:
        """提取疑问词"""
        question_words = ["who", "what", "when", "where", "why", "how", "which"]
        query_lower = query.lower()
        for qw in question_words:
            if qw in query_lower:
                return qw
        return None

    def _check_context_dependency(self, query: str, session_id: str) -> bool:
        """检查是否依赖上下文"""
        # 代词检测
        pronouns = ["it", "that", "this", "them", "those", "he", "she"]
        if any(p in query.lower() for p in pronouns):
            return True

        # 短查询检测
        if len(query.split()) < 5 and session_id:
            return True

        return False

    def _get_complexity(self, intent_type: str) -> str:
        """获取意图复杂度"""
        complexity_map = {
            "factual": "simple",
            "analytical": "medium",
            "comparative": "medium",
            "multi_hop": "high",
            "exploratory": "medium",
            "follow_up": "contextual"
        }
        return complexity_map.get(intent_type, "medium")
```

### 3. Memory Agent（记忆管理Agent）

**角色定位**：管理会话记忆、上下文和历史交互

#### 核心职责

- ✅ 维护会话级别的对话历史
- ✅ 缓存实体、主题、关系等上下文信息
- ✅ 支持跨轮对话的上下文恢复
- ✅ 提供记忆检索和相关性排序

#### 记忆层次结构

```
Memory Agent
├── Short-Term Memory（短期记忆）
│   ├── Current Session（当前会话）
│   │   ├── Query History（查询历史）
│   │   ├── Entity Cache（实体缓存）
│   │   └── Topic Stack（主题栈）
│   └── Recent Sessions（最近会话，最多10个）
│
├── Long-Term Memory（长期记忆）
│   ├── User Profile（用户画像）
│   ├── Frequent Queries（高频查询）
│   └── Domain Knowledge（领域知识）
│
└── Working Memory（工作记忆）
    ├── Current Context（当前上下文）
    └── Intermediate Results（中间结果）
```

#### 关键方法

```python
class MemoryAgent:
    """记忆管理Agent"""

    def __init__(self, message_bus, storage_backend="redis"):
        self.message_bus = message_bus
        self.storage = self._init_storage(storage_backend)
        self.sessions = {}  # 活跃会话缓存

        # 注册消息处理器
        self.message_bus.register_handler("memory_agent", self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "get_context":
            result = await self.get_context(payload["session_id"])
        elif action == "save_interaction":
            result = await self.save_interaction(
                payload["session_id"],
                payload["query"],
                payload["answer"],
                payload.get("intent")
            )
        elif action == "update_cache":
            result = await self.update_cache(
                payload["session_id"],
                payload["cache_type"],
                payload["data"]
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="memory_agent",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def get_context(self, session_id: str) -> Dict:
        """获取会话上下文"""
        # 从缓存或存储加载会话
        session = await self._load_session(session_id)

        if not session:
            return {
                "history": [],
                "entities": [],
                "topics": [],
                "last_intent": None
            }

        return {
            "history": session.get_recent_history(10),
            "entities": session.get_mentioned_entities(),
            "topics": session.get_topics(),
            "last_intent": session.get_last_intent(),
            "user_profile": session.get_user_profile()
        }

    async def save_interaction(self, session_id: str, query: str,
                              answer: Dict, intent: str = None) -> Dict:
        """保存交互到记忆"""
        session = await self._load_session(session_id)

        # 添加到历史
        interaction = {
            "query": query,
            "answer": answer.get("answer", ""),
            "intent": intent,
            "timestamp": time.time(),
            "entities": answer.get("entities", []),
            "retrieval_stats": answer.get("retrieval_stats", {})
        }

        session.add_interaction(interaction)

        # 持久化到存储
        await self._save_session(session_id, session)

        return {"status": "saved", "session_id": session_id}

    async def update_cache(self, session_id: str, cache_type: str,
                          data: Any) -> Dict:
        """更新缓存"""
        session = await self._load_session(session_id)

        if cache_type == "entities":
            session.entity_cache.update(data)
        elif cache_type == "topics":
            session.topic_cache.append(data)
        elif cache_type == "relations":
            session.relation_cache.update(data)

        await self._save_session(session_id, session)

        return {"status": "updated"}

    async def _load_session(self, session_id: str):
        """加载会话"""
        # 先检查内存缓存
        if session_id in self.sessions:
            return self.sessions[session_id]

        # 从持久化存储加载
        session_data = await self.storage.get(f"session:{session_id}")

        if session_data:
            session = SessionMemory.from_dict(session_data)
            self.sessions[session_id] = session
            return session

        # 创建新会话
        session = SessionMemory(session_id)
        self.sessions[session_id] = session
        return session

    async def _save_session(self, session_id: str, session):
        """持久化会话"""
        session_data = session.to_dict()
        await self.storage.set(f"session:{session_id}", session_data, ttl=3600)

    def _init_storage(self, backend: str):
        """初始化存储后端"""
        if backend == "redis":
            return RedisStorage()
        elif backend == "memory":
            return InMemoryStorage()
        else:
            raise ValueError(f"Unknown storage backend: {backend}")


class SessionMemory:
    """会话记忆类"""

    def __init__(self, session_id: str):
        self.session_id = session_id
        self.history = []
        self.entity_cache = set()
        self.topic_cache = []
        self.relation_cache = {}
        self.user_profile = {}

    def add_interaction(self, interaction: Dict):
        """添加交互"""
        self.history.append(interaction)

        # 更新实体缓存
        if "entities" in interaction:
            self.entity_cache.update(interaction["entities"])

        # 更新主题缓存
        if "intent" in interaction:
            self.topic_cache.append(interaction["intent"])

    def get_recent_history(self, n: int = 10) -> List[Dict]:
        """获取最近n条历史"""
        return self.history[-n:]

    def get_mentioned_entities(self) -> List[str]:
        """获取提及的实体"""
        return list(self.entity_cache)

    def get_topics(self) -> List[str]:
        """获取讨论主题"""
        return self.topic_cache[-5:]

    def get_last_intent(self) -> str:
        """获取上一次意图"""
        for item in reversed(self.history):
            if item.get("intent"):
                return item["intent"]
        return None

    def get_user_profile(self) -> Dict:
        """获取用户画像"""
        return self.user_profile

    def to_dict(self) -> Dict:
        """序列化"""
        return {
            "session_id": self.session_id,
            "history": self.history,
            "entity_cache": list(self.entity_cache),
            "topic_cache": self.topic_cache,
            "relation_cache": self.relation_cache,
            "user_profile": self.user_profile
        }

    @classmethod
    def from_dict(cls, data: Dict):
        """反序列化"""
        session = cls(data["session_id"])
        session.history = data["history"]
        session.entity_cache = set(data["entity_cache"])
        session.topic_cache = data["topic_cache"]
        session.relation_cache = data["relation_cache"]
        session.user_profile = data["user_profile"]
        return session
```

### 4. Strategy Agent（策略决策Agent）

**角色定位**：根据意图和上下文选择最优检索和推理策略

#### 核心职责

- ✅ 维护策略库（6种检索策略）
- ✅ 根据意图类型、查询复杂度选择策略
- ✅ 动态调整策略参数（top_k, max_hops等）
- ✅ 根据反馈优化策略选择

#### 策略库

| 策略名称                  | 适用意图    | 描述             | 参数                   |
| ------------------------- | ----------- | ---------------- | ---------------------- |
| **DirectRetrieval**       | factual     | 直接检索，单跳   | `top_k=5`              |
| **MultiHopRetrieval**     | analytical  | 多跳图遍历       | `max_hops=2, top_k=10` |
| **ParallelRetrieval**     | comparative | 并行检索多个实体 | `parallel_count=2`     |
| **IterativeReasoning**    | multi_hop   | 迭代检索+推理    | `max_iterations=3`     |
| **ExpansionRetrieval**    | exploratory | 扩展式检索       | `expansion_factor=1.5` |
| **ContextAwareRetrieval** | follow_up   | 上下文感知检索   | 依赖记忆               |

#### 关键方法

```python
class StrategyAgent:
    """策略决策Agent"""

    def __init__(self, message_bus, config):
        self.message_bus = message_bus
        self.config = config
        self.strategy_bank = self._init_strategy_bank()

        # 注册消息处理器
        self.message_bus.register_handler("strategy_agent", self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "decide_strategy":
            result = await self.decide_strategy(
                payload["intent"],
                payload["context"],
                payload["query"]
            )
        elif action == "adjust_strategy":
            result = await self.adjust_strategy(
                payload["feedback"],
                payload["suggestions"]
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="strategy_agent",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def decide_strategy(self, intent: Dict, context: Dict,
                             query: str) -> Dict:
        """决策检索策略"""
        intent_type = intent["intent_type"]
        complexity = intent["complexity"]

        # 基础策略选择
        base_strategy_name = self._map_intent_to_strategy(intent_type)
        base_strategy = self.strategy_bank[base_strategy_name]

        # 定制策略参数
        customized_strategy = self._customize_strategy_params(
            base_strategy, intent, context
        )

        return {
            "strategy_name": base_strategy_name,
            "strategy_params": customized_strategy.params,
            "estimated_time": self._estimate_time(customized_strategy),
            "estimated_cost": self._estimate_cost(customized_strategy)
        }

    def _map_intent_to_strategy(self, intent_type: str) -> str:
        """意图到策略的映射"""
        mapping = {
            "factual": "DirectRetrieval",
            "analytical": "MultiHopRetrieval",
            "comparative": "ParallelRetrieval",
            "multi_hop": "IterativeReasoning",
            "exploratory": "ExpansionRetrieval",
            "follow_up": "ContextAwareRetrieval"
        }
        return mapping.get(intent_type, "DirectRetrieval")

    def _customize_strategy_params(self, strategy, intent: Dict,
                                   context: Dict):
        """定制策略参数"""
        complexity = intent["complexity"]
        entity_count = intent["features"]["entity_count"]

        # 根据复杂度调整参数
        if complexity == "simple":
            strategy.set_params(
                max_hops=1,
                top_k=5,
                use_reranking=False
            )
        elif complexity == "medium":
            strategy.set_params(
                max_hops=2,
                top_k=10,
                use_reranking=True
            )
        elif complexity == "high":
            strategy.set_params(
                max_hops=3,
                top_k=15,
                use_reranking=True,
                use_reflection=True
            )

        # 根据实体数量调整
        if entity_count > 3:
            strategy.set_params(
                expansion_factor=1.5,
                use_entity_linking=True
            )

        return strategy

    def _estimate_time(self, strategy) -> float:
        """估计执行时间（秒）"""
        base_time = {
            "DirectRetrieval": 0.5,
            "MultiHopRetrieval": 1.5,
            "ParallelRetrieval": 1.0,
            "IterativeReasoning": 3.0,
            "ExpansionRetrieval": 2.0,
            "ContextAwareRetrieval": 0.8
        }
        return base_time.get(strategy.__class__.__name__, 1.0)

    def _estimate_cost(self, strategy) -> float:
        """估计LLM token成本"""
        # 简化版：基于策略复杂度
        return strategy.params.get("max_hops", 1) * 1000  # tokens

    async def adjust_strategy(self, feedback: Dict,
                             suggestions: List[str]) -> Dict:
        """根据反馈调整策略"""
        current_strategy_name = feedback.get("strategy_name")
        current_strategy = self.strategy_bank[current_strategy_name]

        # 解析建议
        if "increase top_k" in str(suggestions).lower():
            current_top_k = current_strategy.params.get("top_k", 5)
            current_strategy.set_params(top_k=min(current_top_k * 2, 30))

        if "expand search" in str(suggestions).lower():
            current_strategy.set_params(expansion_factor=1.5)

        if "increase depth" in str(suggestions).lower():
            current_hops = current_strategy.params.get("max_hops", 1)
            current_strategy.set_params(max_hops=min(current_hops + 1, 5))

        return {
            "strategy_name": current_strategy_name,
            "strategy_params": current_strategy.params,
            "adjusted": True
        }

    def _init_strategy_bank(self) -> Dict:
        """初始化策略库"""
        return {
            "DirectRetrieval": DirectRetrievalStrategy(),
            "MultiHopRetrieval": MultiHopRetrievalStrategy(),
            "ParallelRetrieval": ParallelRetrievalStrategy(),
            "IterativeReasoning": IterativeReasoningStrategy(),
            "ExpansionRetrieval": ExpansionRetrievalStrategy(),
            "ContextAwareRetrieval": ContextAwareRetrievalStrategy()
        }
```

### 5. Retrieval Coordinator Agent（检索协调Agent）

**角色定位**：协调并管理多个并行检索Agent

#### 核心职责

- ✅ 根据策略创建和调度Retrieval Agent
- ✅ 管理Retrieval Agent池（动态扩缩容）
- ✅ 聚合多个Retrieval Agent的结果
- ✅ 触发Reranking Agent进行结果重排

#### 关键方法

```python
class RetrievalCoordinatorAgent:
    """检索协调Agent"""

    def __init__(self, message_bus, kt_retriever, max_parallel_agents=5):
        self.message_bus = message_bus
        self.kt_retriever = kt_retriever
        self.max_parallel_agents = max_parallel_agents
        self.agent_pool = []

        # 注册消息处理器
        self.message_bus.register_handler("retrieval_coordinator",
                                         self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "coordinate_retrieval":
            result = await self.coordinate_retrieval(
                payload["query"],
                payload["strategy"],
                payload.get("intent", {})
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="retrieval_coordinator",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def coordinate_retrieval(self, query: str, strategy: Dict,
                                   intent: Dict) -> Dict:
        """协调检索任务"""
        strategy_name = strategy["strategy_name"]
        strategy_params = strategy["strategy_params"]

        # 根据策略类型选择执行方式
        if strategy_name == "ParallelRetrieval":
            # 并行检索（针对比较性查询）
            result = await self._parallel_retrieval(query, strategy_params, intent)

        elif strategy_name == "IterativeReasoning":
            # 迭代检索（针对多跳推理）
            result = await self._iterative_retrieval(query, strategy_params, intent)

        else:
            # 单次检索
            result = await self._single_retrieval(query, strategy_params)

        # 触发重排序
        if strategy_params.get("use_reranking", False):
            result = await self._trigger_reranking(query, result)

        return result

    async def _parallel_retrieval(self, query: str, params: Dict,
                                 intent: Dict) -> Dict:
        """并行检索多个子查询"""
        # 提取需要并行检索的实体
        entities = intent["features"]["entities"]

        # 创建多个Retrieval Agent
        tasks = []
        for entity in entities[:self.max_parallel_agents]:
            agent = ParallelRetrievalAgent(
                agent_id=f"retrieval_{entity}",
                kt_retriever=self.kt_retriever,
                params=params
            )

            task = agent.retrieve(entity, params.get("top_k", 10))
            tasks.append(task)

        # 并行执行
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 聚合结果
        aggregated_result = self._aggregate_results(results)

        return {
            "results": aggregated_result,
            "strategy": "ParallelRetrieval",
            "agent_count": len(tasks),
            "stats": self._compute_stats(results)
        }

    async def _iterative_retrieval(self, query: str, params: Dict,
                                  intent: Dict) -> Dict:
        """迭代检索（多跳推理）"""
        max_iterations = params.get("max_iterations", 3)
        current_query = query
        all_results = []

        for iteration in range(max_iterations):
            # 单次检索
            iter_result = await self._single_retrieval(current_query, params)
            all_results.append(iter_result)

            # 判断是否需要继续
            if self._should_stop_iteration(iter_result, iteration):
                break

            # 基于当前结果生成下一个查询
            current_query = await self._generate_next_query(
                query, iter_result, iteration
            )

        return {
            "results": all_results,
            "strategy": "IterativeReasoning",
            "iterations": len(all_results),
            "stats": self._compute_stats(all_results)
        }

    async def _single_retrieval(self, query: str, params: Dict) -> Dict:
        """单次检索"""
        agent = SingleRetrievalAgent(
            agent_id="retrieval_single",
            kt_retriever=self.kt_retriever,
            params=params
        )

        result = await agent.retrieve(
            query,
            top_k=params.get("top_k", 10),
            max_hops=params.get("max_hops", 1)
        )

        return result

    async def _trigger_reranking(self, query: str, result: Dict) -> Dict:
        """触发重排序Agent"""
        # 发送消息给Reranking Agent
        message = Message(
            sender="retrieval_coordinator",
            receiver="reranking_agent",
            action="rerank",
            payload={"query": query, "results": result},
            timestamp=time.time()
        )

        response = await self.message_bus.send_and_wait(message, timeout=5.0)
        return response.payload

    def _aggregate_results(self, results: List[Dict]) -> List[Dict]:
        """聚合多个检索结果"""
        # 合并、去重、排序
        all_items = []
        for result in results:
            if isinstance(result, dict) and "items" in result:
                all_items.extend(result["items"])

        # 去重（基于ID）
        unique_items = {item["id"]: item for item in all_items}

        # 排序（基于分数）
        sorted_items = sorted(
            unique_items.values(),
            key=lambda x: x.get("score", 0),
            reverse=True
        )

        return sorted_items

    def _compute_stats(self, results: Union[List, Dict]) -> Dict:
        """计算统计信息"""
        if isinstance(results, dict):
            results = [results]

        total_items = sum(len(r.get("items", [])) for r in results
                         if isinstance(r, dict))

        return {
            "total_items": total_items,
            "result_count": len(results)
        }

    def _should_stop_iteration(self, result: Dict, iteration: int) -> bool:
        """判断是否停止迭代"""
        # 如果结果足够（items数量 > 5）
        if len(result.get("items", [])) >= 5:
            return True

        # 如果已达到最大迭代次数
        if iteration >= 2:
            return True

        return False

    async def _generate_next_query(self, original_query: str,
                                   current_result: Dict,
                                   iteration: int) -> str:
        """生成下一个查询（用于迭代）"""
        # 简化版：基于当前结果提取关键实体
        items = current_result.get("items", [])
        if not items:
            return original_query

        # 提取top结果中的实体
        top_item = items[0]
        new_entities = top_item.get("entities", [])

        # 构造新查询（可以用LLM优化）
        if new_entities:
            return f"{original_query} related to {new_entities[0]}"

        return original_query


class ParallelRetrievalAgent:
    """并行检索Agent（轻量级）"""

    def __init__(self, agent_id: str, kt_retriever, params: Dict):
        self.agent_id = agent_id
        self.kt_retriever = kt_retriever
        self.params = params

            query,

            "agent_id": self.agent_id,
```

### 6. Reasoning Agent（推理规划Agent）

**角色定位**：制定推理计划，执行逻辑推理

#### 核心职责

- ✅ 分析检索结果，制定推理路径
- ✅ 执行逻辑推理（因果、归纳、演绎）
- ✅ 处理多跳推理链
- ✅ 生成推理解释

#### 关键方法

```python
class ReasoningAgent:
    """推理规划Agent"""

    def __init__(self, message_bus, llm_client):
        self.message_bus = message_bus
        self.llm_client = llm_client

        # 注册消息处理器
        self.message_bus.register_handler("reasoning_agent", self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "plan_reasoning":
            result = await self.plan_reasoning(
                payload["query"],
                payload["retrieval_result"],
                payload.get("intent", {})
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="reasoning_agent",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def plan_reasoning(self, query: str, retrieval_result: Dict,
                            intent: Dict) -> Dict:
        """制定推理计划"""
        intent_type = intent.get("intent_type", "factual")

        # 根据意图选择推理方法
        if intent_type == "factual":
            reasoning_result = await self._factual_reasoning(
                query, retrieval_result
            )

        elif intent_type == "analytical":
            reasoning_result = await self._analytical_reasoning(
                query, retrieval_result
            )

        elif intent_type == "comparative":
            reasoning_result = await self._comparative_reasoning(
                query, retrieval_result
            )

        elif intent_type == "multi_hop":
            reasoning_result = await self._multi_hop_reasoning(
                query, retrieval_result
            )

        else:
            reasoning_result = await self._default_reasoning(
                query, retrieval_result
            )

        return reasoning_result

    async def _factual_reasoning(self, query: str,
                                retrieval_result: Dict) -> Dict:
        """事实性推理（简单查找）"""
        items = retrieval_result.get("items", [])

        if not items:
            return {
                "reasoning_type": "factual",
                "answer": "未找到相关信息",
                "confidence": 0.0,
                "evidence": []
            }

        # 直接返回top结果
        top_item = items[0]

        return {
            "reasoning_type": "factual",
            "answer": top_item.get("content", ""),
            "confidence": top_item.get("score", 0.5),
            "evidence": [top_item]
        }

    async def _analytical_reasoning(self, query: str,
                                   retrieval_result: Dict) -> Dict:
        """分析性推理（Why/How）"""
        items = retrieval_result.get("items", [])

        # 构造推理上下文
        context = "\n".join([item.get("content", "") for item in items[:5]])

        # 使用LLM进行因果推理
        prompt = f"""Based on the following information, analyze and explain:

Query: {query}

Context:
{context}

Provide a detailed analytical answer explaining the reasoning.
"""

        answer = await self.llm_client.call_async(prompt, max_tokens=300)

        return {
            "reasoning_type": "analytical",
            "answer": answer,
            "confidence": 0.8,
            "evidence": items[:5],
            "reasoning_chain": ["Extract context", "Causal analysis", "Generate explanation"]
        }

    async def _comparative_reasoning(self, query: str,
                                    retrieval_result: Dict) -> Dict:
        """比较性推理"""
        items = retrieval_result.get("items", [])

        # 假设前半部分是实体A，后半部分是实体B
        mid = len(items) // 2
        entity_a_items = items[:mid]
        entity_b_items = items[mid:]

        # 提取属性
        attributes_a = self._extract_attributes(entity_a_items)
        attributes_b = self._extract_attributes(entity_b_items)

        # 对比
        comparison = self._compare_attributes(attributes_a, attributes_b)

        return {
            "reasoning_type": "comparative",
            "answer": comparison["summary"],
            "confidence": 0.75,
            "evidence": items,
            "comparison_table": comparison["table"]
        }

    async def _multi_hop_reasoning(self, query: str,
                                  retrieval_result: Dict) -> Dict:
        """多跳推理"""
        # 如果检索结果包含多个迭代
        iterations = retrieval_result.get("results", [])

        # 构建推理链
        reasoning_chain = []
        for i, iter_result in enumerate(iterations):
            step = {
                "step": i + 1,
                "sub_question": iter_result.get("query", query),
                "answer": iter_result.get("items", [])[0].get("content", "")
                         if iter_result.get("items") else "",
                "evidence_count": len(iter_result.get("items", []))
            }
            reasoning_chain.append(step)

        # 综合答案
        final_answer = await self._synthesize_multi_hop_answer(
            query, reasoning_chain
        )

        return {
            "reasoning_type": "multi_hop",
            "answer": final_answer,
            "confidence": 0.7,
            "reasoning_chain": reasoning_chain,
            "hops": len(reasoning_chain)
        }

    async def _default_reasoning(self, query: str,
                                retrieval_result: Dict) -> Dict:
        """默认推理"""
        items = retrieval_result.get("items", [])
        context = "\n".join([item.get("content", "") for item in items[:3]])

        prompt = f"""Answer the following query based on the context:

Query: {query}

Context:
{context}

Answer:
"""

        answer = await self.llm_client.call_async(prompt, max_tokens=200)

        return {
            "reasoning_type": "default",
            "answer": answer,
            "confidence": 0.6,
            "evidence": items[:3]
        }

    def _extract_attributes(self, items: List[Dict]) -> Dict:
        """提取实体属性"""
        attributes = {}
        for item in items:
            # 假设item包含(entity, attribute, value)信息
            if "attributes" in item:
                attributes.update(item["attributes"])
        return attributes

    def _compare_attributes(self, attr_a: Dict, attr_b: Dict) -> Dict:
        """比较属性"""
        comparison_table = []
        all_keys = set(attr_a.keys()) | set(attr_b.keys())

        for key in all_keys:
            comparison_table.append({
                "attribute": key,
                "entity_a": attr_a.get(key, "N/A"),
                "entity_b": attr_b.get(key, "N/A"),
                "same": attr_a.get(key) == attr_b.get(key)
            })

        summary = f"Comparison of {len(all_keys)} attributes"

        return {
            "table": comparison_table,
            "summary": summary
        }

    async def _synthesize_multi_hop_answer(self, query: str,
                                          reasoning_chain: List[Dict]) -> str:
        """综合多跳推理答案"""
        chain_text = "\n".join([
            f"Step {step['step']}: {step['sub_question']} → {step['answer']}"
            for step in reasoning_chain
        ])

        prompt = f"""Based on the following reasoning chain, provide a final answer to the query:

Query: {query}

Reasoning Chain:
{chain_text}

Final Answer:
"""

        answer = await self.llm_client.call_async(prompt, max_tokens=200)
        return answer
```

### 7. Reflection Agent（反思评估Agent）

**角色定位**：评估答案质量，提供改进建议

#### 核心职责

- ✅ 评估答案的质量（准确性、完整性、相关性）
- ✅ 识别答案中的潜在问题
- ✅ 生成改进建议
- ✅ 决定是否需要重试

#### 关键方法

```python
class ReflectionAgent:
    """反思评估Agent"""

    def __init__(self, message_bus, llm_client, quality_threshold=0.8):
        self.message_bus = message_bus
        self.llm_client = llm_client
        self.quality_threshold = quality_threshold

        # 注册消息处理器
        self.message_bus.register_handler("reflection_agent", self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "evaluate_quality":
            result = await self.evaluate_quality(
                payload["query"],
                payload["answer"],
                payload.get("retrieval", {})
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="reflection_agent",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def evaluate_quality(self, query: str, answer: Dict,
                              retrieval_result: Dict) -> Dict:
        """评估答案质量"""
        # 多维度评估
        scores = {}

        # 1. 相关性评分
        scores["relevance"] = await self._evaluate_relevance(query, answer)

        # 2. 完整性评分
        scores["completeness"] = await self._evaluate_completeness(
            query, answer, retrieval_result
        )

        # 3. 准确性评分（如果有ground truth）
        scores["accuracy"] = await self._evaluate_accuracy(query, answer)

        # 4. 连贯性评分
        scores["coherence"] = await self._evaluate_coherence(answer)

        # 综合质量分数
        quality_score = sum(scores.values()) / len(scores)

        # 识别问题
        issues = self._identify_issues(scores, quality_score)

        # 生成建议
        suggestions = self._generate_suggestions(issues, scores)

        # 决定是否重试
        should_retry = quality_score < self.quality_threshold and \
                      len(issues) > 0

        return {
            "quality_score": quality_score,
            "scores": scores,
            "issues": issues,
            "suggestions": suggestions,
            "should_retry": should_retry,
            "retry_count": 0
        }

    async def _evaluate_relevance(self, query: str, answer: Dict) -> float:
        """评估相关性"""
        from sentence_transformers import SentenceTransformer
        model = SentenceTransformer('all-MiniLM-L6-v2')

        query_emb = model.encode([query])
        answer_text = answer.get("answer", "")
        answer_emb = model.encode([answer_text])

        from sklearn.metrics.pairwise import cosine_similarity
        similarity = cosine_similarity(query_emb, answer_emb)[0][0]

        return float(similarity)

    async def _evaluate_completeness(self, query: str, answer: Dict,
                                    retrieval_result: Dict) -> float:
        """评估完整性"""
        # 提取查询中的关键实体
        import spacy
        nlp = spacy.load("en_core_web_sm")
        doc = nlp(query)
        key_entities = [ent.text for ent in doc.ents]

        if not key_entities:
            return 0.7  # 默认分数

        # 检查答案中是否覆盖了关键实体
        answer_text = answer.get("answer", "").lower()
        covered = sum(1 for ent in key_entities if ent.lower() in answer_text)

        completeness = covered / len(key_entities)
        return completeness

    async def _evaluate_accuracy(self, query: str, answer: Dict) -> float:
        """评估准确性（使用LLM判断）"""
        # 简化版：使用LLM进行自我评估
        prompt = f"""Evaluate the accuracy of the following answer to the query.
Rate on a scale of 0.0 to 1.0.

Query: {query}
Answer: {answer.get('answer', '')}

Accuracy score (0.0-1.0):
"""

        response = await self.llm_client.call_async(prompt, max_tokens=10)

        try:
            score = float(response.strip())
            return min(max(score, 0.0), 1.0)
        except:
            return 0.5  # 默认分数

    async def _evaluate_coherence(self, answer: Dict) -> float:
        """评估连贯性"""
        answer_text = answer.get("answer", "")

        # 简化版：检查基本指标
        if len(answer_text) < 10:
            return 0.3

        # 检查是否有断句
        if "." in answer_text or "。" in answer_text:
            return 0.8

        return 0.6

    def _identify_issues(self, scores: Dict, overall_score: float) -> List[str]:
        """识别问题"""
        issues = []

        if scores["relevance"] < 0.6:
            issues.append("Low relevance: Answer may not address the query")

        if scores["completeness"] < 0.5:
            issues.append("Incomplete: Missing key entities or information")

        if scores["accuracy"] < 0.6:
            issues.append("Low accuracy: Answer may contain errors")

        if scores["coherence"] < 0.5:
            issues.append("Poor coherence: Answer is not well-structured")

        if overall_score < 0.5:
            issues.append("Overall quality is below acceptable threshold")

        return issues

    def _generate_suggestions(self, issues: List[str],
                             scores: Dict) -> List[str]:
        """生成改进建议"""
        suggestions = []

        if "Low relevance" in str(issues):
            suggestions.append("Refine retrieval to focus on query keywords")

        if "Incomplete" in str(issues):
            suggestions.append("Increase top_k or expand search scope")

        if "Low accuracy" in str(issues):
            suggestions.append("Use different retrieval strategy or increase depth")

        if "Poor coherence" in str(issues):
            suggestions.append("Improve answer generation prompt")

        return suggestions
```

### 8. Response Generator Agent（答案生成Agent）

**角色定位**：基于推理结果生成最终答案

#### 核心职责

- ✅ 综合检索和推理结果
- ✅ 使用LLM生成自然语言答案
- ✅ 格式化答案输出
- ✅ 添加引用和来源

#### 关键方法

```python
class ResponseGeneratorAgent:
    """答案生成Agent"""

    def __init__(self, message_bus, llm_client):
        self.message_bus = message_bus
        self.llm_client = llm_client

        # 注册消息处理器
        self.message_bus.register_handler("response_generator",
                                         self.handle_message)

    async def handle_message(self, message: Message) -> Message:
        """处理消息"""
        action = message.action
        payload = message.payload

        if action == "generate_answer":
            result = await self.generate_answer(
                payload["query"],
                payload.get("reasoning", {}),
                payload.get("retrieval", {})
            )
        else:
            result = {"error": f"Unknown action: {action}"}

        return Message(
            sender="response_generator",
            receiver=message.sender,
            action=f"{action}_response",
            payload=result,
            timestamp=time.time()
        )

    async def generate_answer(self, query: str, reasoning: Dict,
                             retrieval: Dict) -> Dict:
        """生成最终答案"""
        # 提取推理答案
        reasoning_answer = reasoning.get("answer", "")
        reasoning_type = reasoning.get("reasoning_type", "default")
        evidence = reasoning.get("evidence", [])

        # 构造生成提示
        prompt = self._build_generation_prompt(
            query, reasoning_answer, reasoning_type, evidence
        )

        # 使用LLM生成答案
        generated_answer = await self.llm_client.call_async(
            prompt,
            max_tokens=400,
            temperature=0.7
        )

        # 后处理
        formatted_answer = self._format_answer(generated_answer, evidence)

        return {
            "answer": formatted_answer,
            "raw_answer": generated_answer,
            "reasoning_type": reasoning_type,
            "evidence_count": len(evidence),
            "citations": self._extract_citations(evidence)
        }

    def _build_generation_prompt(self, query: str, reasoning_answer: str,
                                 reasoning_type: str, evidence: List) -> str:
        """构造生成提示"""
        evidence_text = "\n".join([
            f"[{i+1}] {ev.get('content', '')[:200]}"
            for i, ev in enumerate(evidence[:5])
        ])

        prompt = f"""Based on the following reasoning result and evidence, generate a comprehensive answer to the query.

Query: {query}

Reasoning Type: {reasoning_type}
Reasoning Result: {reasoning_answer}

Supporting Evidence:
{evidence_text}

Generate a well-structured, informative answer that:
1. Directly addresses the query
2. Incorporates key information from evidence
3. Is clear and concise
4. Cites sources when appropriate

Answer:
"""

        return prompt

    def _format_answer(self, answer: str, evidence: List) -> str:
        """格式化答案"""
        # 添加引用标记
        formatted = answer.strip()

        # 如果答案较长，添加段落分隔
        if len(formatted) > 300:
            # 简单的段落分隔（每100字加一个换行）
            formatted = "\n\n".join([
                formatted[i:i+100]
                for i in range(0, len(formatted), 100)
            ])

        return formatted

    def _extract_citations(self, evidence: List) -> List[Dict]:
        """提取引用"""
        citations = []
        for i, ev in enumerate(evidence[:5]):
            citations.append({
                "id": i + 1,
                "source": ev.get("source", "Unknown"),
                "preview": ev.get("content", "")[:100] + "..."
            })
        return citations
```

## 📡 消息总线（Message Bus）设计

### 核心功能

- ✅ Agent间的异步通信
- ✅ 消息路由和分发
- ✅ 消息持久化（可选）
- ✅ 超时处理

### 实现

```python
class Message:
    """消息类"""

    def __init__(self, sender: str, receiver: str, action: str,
                 payload: Dict, timestamp: float = None,
                 message_id: str = None):
        self.sender = sender
        self.receiver = receiver
        self.action = action
        self.payload = payload
        self.timestamp = timestamp or time.time()
        self.message_id = message_id or self._generate_id()

    def _generate_id(self) -> str:
        """生成唯一消息ID"""
        import uuid
        return str(uuid.uuid4())

    def to_dict(self) -> Dict:
        """序列化"""
        return {
            "sender": self.sender,
            "receiver": self.receiver,
            "action": self.action,
            "payload": self.payload,
            "timestamp": self.timestamp,
            "message_id": self.message_id
        }

    @classmethod
    def from_dict(cls, data: Dict):
        """反序列化"""
        return cls(
            sender=data["sender"],
            receiver=data["receiver"],
            action=data["action"],
            payload=data["payload"],
            timestamp=data.get("timestamp"),
            message_id=data.get("message_id")
        )


class MessageBus:
    """消息总线"""

    def __init__(self):
        self.handlers = {}  # agent_name -> handler_function
        self.pending_responses = {}  # message_id -> Future
        self.message_queue = asyncio.Queue()

        # 启动消息处理循环
        asyncio.create_task(self._process_messages())

    def register_handler(self, agent_name: str, handler):
        """注册Agent的消息处理器"""
        self.handlers[agent_name] = handler
        logger.info(f"Registered handler for agent: {agent_name}")

    async def send(self, message: Message):
        """发送消息（异步，不等待响应）"""
        await self.message_queue.put(message)

    async def send_and_wait(self, message: Message,
                           timeout: float = 30.0) -> Message:
        """发送消息并等待响应"""
        # 创建Future用于等待响应
        future = asyncio.Future()
        self.pending_responses[message.message_id] = future

        # 发送消息
        await self.message_queue.put(message)

        # 等待响应（带超时）
        try:
            response = await asyncio.wait_for(future, timeout=timeout)
            return response
        except asyncio.TimeoutError:
            logger.error(f"Message {message.message_id} timed out")
            del self.pending_responses[message.message_id]
            raise TimeoutError(f"Message to {message.receiver} timed out")

    async def _process_messages(self):
        """处理消息队列"""
        while True:
            # 从队列获取消息
            message = await self.message_queue.get()

            # 路由到对应的Agent
            receiver = message.receiver

            if receiver not in self.handlers:
                logger.error(f"No handler for agent: {receiver}")
                self.message_queue.task_done()
                continue

            # 调用Handler
            try:
                handler = self.handlers[receiver]
                response = await handler(message)

                # 如果有等待此消息响应的Future，设置结果
                if message.message_id in self.pending_responses:
                    future = self.pending_responses.pop(message.message_id)
                    future.set_result(response)

            except Exception as e:
                logger.error(f"Error processing message: {e}")
                if message.message_id in self.pending_responses:
                    future = self.pending_responses.pop(message.message_id)
                    future.set_exception(e)

            self.message_queue.task_done()
```
