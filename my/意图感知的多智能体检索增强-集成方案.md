## 🎯 集成目标

### 设计原则

1. **向后兼容**：保留现有的`agent`和`noagent`模式
2. **新增模式**：新增`multi-agent`模式
3. **模块解耦**：多Agent系统独立于现有代码
4. **渐进集成**：可以逐步迁移功能到多Agent架构
5. **配置驱动**：通过配置文件切换模式

## 🔗 与现有模块的集成关系

### 集成映射表

| 现有模块                            | 对应的Multi-Agent组件   | 集成方式                                |
| ----------------------------------- | ----------------------- | --------------------------------------- |
| `agentic_decomposer.GraphQ`         | `Reasoning Agent`       | 借鉴查询分解逻辑，封装到Reasoning Agent |
| `enhanced_kt_retriever.KTRetriever` | `Retrieval Coordinator` | 直接调用现有KTRetriever执行检索         |
| `kt_gen.KTBuilder`                  | 保持独立                | 构建阶段不涉及多Agent                   |
| `call_llm_api.LLMCompletionCall`    | 所有Agent共享           | 所有Agent通过统一LLM客户端调用          |
| `tree_comm.FastTreeComm`            | 保持独立                | 社区检测在构建阶段使用                  |

### 集成示例：Retrieval Agent调用现有KTRetriever

```python
# models/agents/retrieval_coordinator.py

from models.retriever.enhanced_kt_retriever import KTRetriever

class RetrievalCoordinatorAgent:
    def __init__(self, message_bus, dataset: str, config):
        self.message_bus = message_bus

        # 直接使用现有的KTRetriever
        self.kt_retriever = KTRetriever(
            dataset=dataset,
            mode="agent",
            config=config
        )

    async def _single_retrieval(self, query: str, params: Dict) -> Dict:
        """调用现有KTRetriever执行检索"""
        result = await asyncio.to_thread(
            self.kt_retriever.retrieve_nodes,
            query,
            params.get("top_k", 10)
        )
        return {"items": result}
```

```python
# core/message_bus.py
import asyncio
from typing import Dict, Callable
from .message import Message
from utils.logger import logger

class MessageBus:
    """异步消息总线"""
    def __init__(self, queue_size: int = 1000):
        self.handlers: Dict[str, Callable] = {}
        self.pending_responses: Dict[str, asyncio.Future] = {}
        self.message_queue = asyncio.Queue(maxsize=queue_size)
        asyncio.create_task(self._process_messages())

    def register_handler(self, agent_name: str, handler: Callable):
        """注册Agent的消息处理器"""
        self.handlers[agent_name] = handler
        logger.info(f"Registered handler for agent: {agent_name}")

    async def send_and_wait(self, message: Message,
                           timeout: float = 30.0) -> Message:
        """发送消息并等待响应"""
        future = asyncio.Future()
        self.pending_responses[message.message_id] = future
        await self.message_queue.put(message)

        try:
            response = await asyncio.wait_for(future, timeout=timeout)
            return response
        except asyncio.TimeoutError:
            logger.error(f"Message {message.message_id} timed out")
            del self.pending_responses[message.message_id]
            raise TimeoutError(f"Message to {message.receiver} timed out")

    async def _process_messages(self):
        """处理消息队列"""
        while True:
            message = await self.message_queue.get()
            receiver = message.receiver

            if receiver not in self.handlers:
                logger.error(f"No handler for agent: {receiver}")
                self.message_queue.task_done()
                continue

            try:
                handler = self.handlers[receiver]
                response = await handler(message)

                if message.message_id in self.pending_responses:
                    future = self.pending_responses.pop(message.message_id)
                    future.set_result(response)
            except Exception as e:
                logger.error(f"Error processing message: {e}")
                if message.message_id in self.pending_responses:
                    future = self.pending_responses.pop(message.message_id)
                    future.set_exception(e)

            self.message_queue.task_done()
```

# models/agents/base_agent.py

from abc import ABC, abstractmethod
from typing import Dict
from core.message import Message

class BaseAgent(ABC):
    """Agent基类"""

    def __init__(self, agent_name: str, message_bus):
        self.agent_name = agent_name
        self.message_bus = message_bus

        # 注册消息处理器
        self.message_bus.register_handler(agent_name, self.handle_message)

    @abstractmethod
    async def handle_message(self, message: Message) -> Message:
        """处理消息（子类必须实现）"""
        pass

    async def send_to_agent(self, receiver: str, action: str,
                           payload: Dict, timeout: float = 30.0) -> Dict:
        """发送消息给其他Agent"""
        message = Message(
            sender=self.agent_name,
            receiver=receiver,
            action=action,
            payload=payload
        )
        response = await self.message_bus.send_and_wait(message, timeout)
        return response.payload
```

from abc import ABC, abstractmethod
from typing import Dict, Any

class RetrievalStrategy(ABC):
    """检索策略基类"""

    def __init__(self):
        self.params = {}

    def set_params(self, **kwargs):
        """设置策略参数"""
        self.params.update(kwargs)

    @abstractmethod
    async def execute(self, query: str, kt_retriever, **kwargs) -> Dict:
        """执行检索（子类实现）"""
        pass
```

from .base_strategy import RetrievalStrategy

class DirectRetrievalStrategy(RetrievalStrategy):
    """直接检索策略"""

        top_k = self.params.get("top_k", 5)
        result = kt_retriever.retrieve_nodes(query, top_k)

### Phase 3: Orchestrator与主流程（Week 6）

#### Step 3.1: 实现Orchestrator

```python
# models/agents/orchestrator.py

from core.message import Message
from utils.logger import logger
import time

class OrchestratorAgent:
    """协调者Agent"""

    def __init__(self, message_bus, config):
        self.message_bus = message_bus
        self.config = config
        self.agents = {}

    def register_agent(self, agent_name: str, agent_instance):
        """注册Agent"""
        self.agents[agent_name] = agent_instance
        logger.info(f"Registered agent: {agent_name}")

    async def process_query(self, query: str, user_id: str = None) -> Dict:
        """处理用户查询（主流程）"""
        session_id = f"session_{int(time.time())}"

        try:
            # Step 1: 意图识别
            intent_result = await self.call_agent(
                "intent_agent",
                action="analyze_intent",
                payload={"query": query, "session_id": session_id}
            )

            # Step 2: 记忆检索
            memory_result = await self.call_agent(
                "memory_agent",
                action="get_context",
                payload={"session_id": session_id}
            )

            # Step 3: 策略决策
            strategy_result = await self.call_agent(
                "strategy_agent",
                action="decide_strategy",
                payload={
                    "intent": intent_result,
                    "context": memory_result,
                    "query": query
                }
            )

            # Step 4: 检索协调
            retrieval_result = await self.call_agent(
                "retrieval_coordinator",
                action="coordinate_retrieval",
                payload={
                    "query": query,
                    "strategy": strategy_result,
                    "intent": intent_result
                }
            )

            # Step 5: 推理规划
            reasoning_result = await self.call_agent(
                "reasoning_agent",
                action="plan_reasoning",
                payload={
                    "query": query,
                    "retrieval_result": retrieval_result,
                    "intent": intent_result
                }
            )

            # Step 6: 答案生成
            response_result = await self.call_agent(
                "response_generator",
                action="generate_answer",
                payload={
                    "query": query,
                    "reasoning": reasoning_result,
                    "retrieval": retrieval_result
                }
            )

            # Step 7: 质量评估
            reflection_result = await self.call_agent(
                "reflection_agent",
                action="evaluate_quality",
                payload={
                    "query": query,
                    "answer": response_result,
                    "retrieval": retrieval_result
                }
            )

            # Step 8: 保存记忆
            await self.call_agent(
                "memory_agent",
                action="save_interaction",
                payload={
                    "session_id": session_id,
                    "query": query,
                    "answer": response_result,
                    "intent": intent_result
                }
            )

            return {
                "answer": response_result["answer"],
                "intent": intent_result["intent_type"],
                "strategy": strategy_result["strategy_name"],
                "quality_score": reflection_result["quality_score"],
                "session_id": session_id
            }

        except Exception as e:
            logger.error(f"Error in orchestrator: {e}")
            return {
                "answer": "系统遇到错误，请稍后重试。",
                "error": str(e)
            }

    async def call_agent(self, agent_name: str, action: str,
                        payload: Dict, timeout: float = 30.0) -> Dict:
        """调用Agent"""
        message = Message(
            sender="orchestrator",
            receiver=agent_name,
            action=action,
            payload=payload
        )
        response = await self.message_bus.send_and_wait(message, timeout)
        return response.payload
```

```python
# main_multi_agent.py

import asyncio
from config import get_config
from core.message_bus import MessageBus
from models.agents.orchestrator import OrchestratorAgent
from models.agents.intent_agent import IntentAgent
from models.agents.memory_agent import MemoryAgent
from models.agents.strategy_agent import StrategyAgent
from models.agents.retrieval_coordinator import RetrievalCoordinatorAgent
from models.agents.reasoning_agent import ReasoningAgent
from models.agents.reflection_agent import ReflectionAgent
from models.agents.response_generator import ResponseGeneratorAgent
from utils.call_llm_api import LLMCompletionCall
from utils.logger import logger

async def initialize_multi_agent_system(dataset: str = "hotpotqa"):
    """初始化多Agent系统"""

    # 加载配置
    config = get_config()
    ma_config = config.get_multi_agent_config()

    if not ma_config or not ma_config.enabled:
        raise ValueError("Multi-agent mode is not enabled in config")

    # 初始化消息总线
    message_bus = MessageBus(queue_size=ma_config.message_queue_size)
    logger.info("Message Bus initialized")

    # 初始化LLM客户端
    llm_client = LLMCompletionCall()
    logger.info("LLM Client initialized")

    # 创建所有Agent
    intent_agent = IntentAgent(message_bus, llm_client)
    memory_agent = MemoryAgent(message_bus, storage_backend=ma_config.storage_backend)
    strategy_agent = StrategyAgent(message_bus, config)
    retrieval_coordinator = RetrievalCoordinatorAgent(
        message_bus, dataset, config, max_parallel_agents=5
    )
    reasoning_agent = ReasoningAgent(message_bus, llm_client)
    reflection_agent = ReflectionAgent(
        message_bus, llm_client, quality_threshold=ma_config.quality_threshold
    )
    response_generator = ResponseGeneratorAgent(message_bus, llm_client)

    logger.info("All agents initialized")

    # 创建Orchestrator
    orchestrator = OrchestratorAgent(message_bus, config)

    # 注册所有Agent
    orchestrator.register_agent("intent_agent", intent_agent)
    orchestrator.register_agent("memory_agent", memory_agent)
    orchestrator.register_agent("strategy_agent", strategy_agent)
    orchestrator.register_agent("retrieval_coordinator", retrieval_coordinator)
    orchestrator.register_agent("reasoning_agent", reasoning_agent)
    orchestrator.register_agent("reflection_agent", reflection_agent)
    orchestrator.register_agent("response_generator", response_generator)

    logger.info("Orchestrator initialized and agents registered")

    return orchestrator

async def main():
    """主函数"""
    import argparse

    parser = argparse.ArgumentParser(description="Multi-Agent RAG System")
    parser.add_argument("--dataset", type=str, default="hotpotqa",
                       help="Dataset name")
    parser.add_argument("--query", type=str, required=True,
                       help="Query to process")
    parser.add_argument("--user_id", type=str, default=None,
                       help="User ID (optional)")

    args = parser.parse_args()

    # 初始化系统
    logger.info(f"Initializing Multi-Agent System for dataset: {args.dataset}")
    orchestrator = await initialize_multi_agent_system(args.dataset)

    # 处理查询
    logger.info(f"Processing query: {args.query}")
    result = await orchestrator.process_query(args.query, args.user_id)

    # 打印结果
    print("\n" + "="*80)
    print(f"Query: {args.query}")
    print(f"Answer: {result['answer']}")
    print(f"Intent: {result.get('intent', 'N/A')}")
    print(f"Strategy: {result.get('strategy', 'N/A')}")
    print(f"Quality Score: {result.get('quality_score', 'N/A')}")
    print(f"Session ID: {result.get('session_id', 'N/A')}")
    print("="*80 + "\n")

if __name__ == "__main__":
    asyncio.run(main())
```
